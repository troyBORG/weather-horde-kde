cmake_minimum_required(VERSION 3.20)
project(weather_plasma_harness LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets Quick)

find_package(KF6Package REQUIRED)
add_executable(plasma-harness main.cpp)
target_link_libraries(plasma-harness PRIVATE Qt6::Widgets Qt6::Quick Plasma::Plasma Plasma::PlasmaQuick KF6::Package)
